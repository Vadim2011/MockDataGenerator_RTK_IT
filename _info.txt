Создание и наполнение БД______________________________________________
Подключитесь к PostgreSQL от имени суперпользователя
sudo -u postgres psql
password: 2******************$

Создание роли (пользователя):
CREATE ROLE mdg WITH LOGIN PASSWORD '2**************x' CREATEDB;

Проверяем добавление пользователя \du

Переход в роль mdg, создание БД:
SET ROLE mdg;
CREATE DATABASE mdg_db;
Проверяем добавление ДБ \l

Переходим в БД mdg_db, создаем таблицы БД, устанавливаем PK, FK, накладываем ограничения на атрибуты:
\c mdg_db;


CREATE TABLE users_ (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(16) NOT NULL
);

CREATE TABLE user_connections (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_user INT NOT NULL,
    time_connect TIMESTAMP WITHOUT TIME ZONE DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_user FOREIGN KEY (id_user) REFERENCES users_ (id) ON DELETE CASCADE
);


CREATE TABLE cities (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(32) NOT NULL
);

CREATE TABLE street (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(32) NOT NULL
);


CREATE TABLE first_name (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(32) NOT NULL
);

CREATE TABLE surname (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(32) NOT NULL
);

CREATE TABLE patronymic (
    id INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR(32) NOT NULL
);
